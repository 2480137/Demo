package org.example;
import java.sql.*;

public class App {
    public static void main(String[] args) throws SQLException, ClassNotFoundException {
        System.out.println("JDBC sample");


        // Register driver
//        Class.forName("com.mysql.cj.jdbc.Driver");

        // connection establish
        Connection connection = DriverManager.getConnection("jdbc:mysql://localhost:3306/demo", "root", "root");
        System.out.println(connection);

        // create statement channel
        Statement statement = connection.createStatement();

        int rowsUpdated = statement.executeUpdate(
                "UPDATE Person SET Address='Chennai', City='Tamil Nadu' WHERE PersonID = 2"
        );

        System.out.println("Rows updated: " + rowsUpdated);


// INSERT
        statement.executeUpdate(
                "INSERT INTO Person (FirstName, LastName, DateOfBirth, Gender, Address, City) " +
                        "VALUES ('siva', 'nad', '1990-05-15', 'M', 'Park Street', 'New York')"
        );

        // INSERT
        statement.executeUpdate(
                "INSERT INTO Person (FirstName, LastName, DateOfBirth, Gender, Address, City) " +
                        "VALUES ('siva', 'test', '1990-05-15', 'M', 'Park Street', 'New York')"
        );


// DELETE
        statement.executeUpdate(
                "DELETE FROM Person WHERE PersonID = 4"
        );


        //view data
        ResultSet resultSet =statement.executeQuery("Select personID,FirstName, Gender,Address from Person");

        while(resultSet.next()) {
            System.out.println(resultSet.getString("personID")+":"+resultSet.getString("FirstName")
            +":"+resultSet.getString("Address"));
        }

        connection.close();

    }
}
